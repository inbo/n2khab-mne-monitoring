## extract `unit_code` from `notes` in #TargetPoints 

*This was a simple thing, until the copy-paste issue when copying from obsidian caused me to delete all notes AND unit_codes on #production.
I could restore this by a ninja update. Code below is safer.*

Was applied to #locevaldb (schema `inbound`) and #mnmsurfdb (schema `transfer`).

```sql
\COPY (
SELECT DISTINCT targetpoint_id, unit_code, wkb_geometry
FROM "inbound"."TargetPoints"
WHERE is_legacypoint AND unit_code IS NOT NULL
ORDER BY targetpoint_id ASC
) TO '/data/mnm_db_backups/20261116_targetpoint_unitcodes_lost.csv' With CSV DELIMITER ',' HEADER
;

BEGIN;
ALTER TABLE "inbound"."TargetPoints" ADD COLUMN unit_code varchar;
COMMENT ON COLUMN "inbound"."TargetPoints".unit_code IS E'location code from upstream sources';

UPDATE "inbound"."TargetPoints"
SET unit_code = notes
WHERE is_legacypoint;

-- SELECT DISTINCT notes, unit_code
-- FROM "inbound"."TargetPoints"
-- WHERE is_legacypoint;

UPDATE "inbound"."TargetPoints"
SET notes = NULL
WHERE is_legacypoint
AND unit_code IS NOT NULL
;

COMMIT;
```

## rename #LocationAssessments to #OrthophotoEvaluations
... because that is what they really are.
Attention: there is a view called `OrthophotoAssessment`; kept for legacy compatibility but probably confusing.

```sql
BEGIN;

ALTER TABLE "outbound"."LocationAssessments"
RENAME TO "OrthophotoEvaluations";

ALTER TABLE "outbound"."OrthophotoEvaluations"
RENAME locationassessment_id TO orthophotoevaluation_id;

-- update views: OrthophotoAssessment, LocationAssessments and gwTransfer
DROP VIEW IF EXISTS  "outbound"."LocationAssessments" ;
CREATE VIEW  "outbound"."LocationAssessments"  AS
  SELECT  orthophotoevaluation_id AS locationassessment_id, *
  FROM  "outbound"."OrthophotoEvaluations";
GRANT SELECT ON  "outbound"."LocationAssessments"  TO viewer_mnmdb;
GRANT UPDATE ON  "outbound"."LocationAssessments"  TO user_loceval;


DROP TRIGGER IF EXISTS log_orthophotoevaluations ON "outbound"."OrthophotoEvaluations";
CREATE TRIGGER log_orthophotoevaluations
BEFORE UPDATE ON "outbound"."OrthophotoEvaluations"
FOR EACH ROW EXECUTE PROCEDURE "metadata".sync_mod();

COMMIT;

```
