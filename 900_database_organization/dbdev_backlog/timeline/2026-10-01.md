## inconsistent use of `stratum` in `type` columns due to #locevaldb bug

*Consequence of [[aquatic habitat types entered locevaldb as stratum instead of type and thereby created SampleUnits duplicates]]*;
issue for correct presentation in QGIS projects.

+ convert implicit stratum to type in aggregated column `type_subset` of #LocationJournals
    + all databases: #mnmsyncdb, #locevaldb, #mnmgwdb, #mnmsurfdb
	+ ```sql
	BEGIN;
	
	WITH unnested_types AS (
	  SELECT
		locationjournal_id AS lojo_id,
		STRING_TO_ARRAY(
		  TRIM(UNNEST(
			STRING_TO_ARRAY(type_subset, ',')
		  )),
		  '_'
		)
		AS stratum_arr
	  FROM "outbound"."LocationJournals"
	  ORDER BY locationjournal_id ASC
	)
	UPDATE "outbound"."LocationJournals" AS TRGTAB
	  SET type_subset = SRCTAB.type_subset
	FROM
	(
	  SELECT DISTINCT lojo_id, STRING_AGG(types, ',') AS type_subset
	  FROM (
		SELECT
		  lojo_id,
		  stratum_arr,
		  ARRAY_TO_STRING(CASE
			WHEN CARDINALITY(stratum_arr) = 3
			THEN stratum_arr[1:1]
		  ELSE
			CASE WHEN CARDINALITY(stratum_arr) > 3
			THEN stratum_arr[1:2]
			ELSE stratum_arr
			END
		  END, '_') AS types
		FROM unnested_types
	  )
	  GROUP BY lojo_id
	) AS SRCTAB
	WHERE
	 (TRGTAB.locationjournal_id = SRCTAB.lojo_id)
	;
	
	COMMIT;
	
	
	SELECT DISTINCT type_subset
	FROM "outbound"."LocationJournals"
	;
	
	```

+ convert implicit stratum to type in column `type` of #LocationEvaluations
    + all affected databases: #mnmgwdb, #mnmsurfdb
    + Should have no affect on `mnmgwdb`
	+ ```sql
    BEGIN;

    WITH unnested_types AS (
      SELECT
      locationevaluation_id AS le_id,
      STRING_TO_ARRAY(type, '_') AS stratum_arr
      FROM "transfer"."LocationEvaluations"
      ORDER BY locationevaluation_id ASC
    )
    UPDATE "transfer"."LocationEvaluations" AS TRGTAB
      SET type = SRCTAB.type
    FROM
    (
      SELECT
        le_id,
        stratum_arr,
        ARRAY_TO_STRING(CASE
        WHEN CARDINALITY(stratum_arr) = 3
        THEN stratum_arr[1:1]
        ELSE
        CASE WHEN CARDINALITY(stratum_arr) > 3
        THEN stratum_arr[1:2]
        ELSE stratum_arr
        END
        END, '_') AS type
      FROM unnested_types
    ) AS SRCTAB
    WHERE
     (TRGTAB.locationevaluation_id = SRCTAB.le_id)
    ;
  
    COMMIT;
  
    SELECT DISTINCT type
    FROM "transfer"."LocationEvaluations"
    ;
 
    ```
