### #drop table #SSPSTaPas

```sql
BEGIN;

DROP VIEW IF EXISTS "inbound"."AllVisits" CASCADE;

ALTER TABLE "inbound"."Visits" DROP CONSTRAINT fk_sspstapas_visits;
ALTER TABLE "inbound"."Visits" DROP COLUMN sspstapa_id;

ALTER TABLE "outbound"."FieldCalendars" DROP CONSTRAINT fk_sspstapas_fieldcalendars;
ALTER TABLE "outbound"."FieldCalendars" DROP COLUMN sspstapa_id;


DROP TABLE "metadata"."SSPSTaPas";


CREATE VIEW "inbound"."AllVisits" AS
SELECT *
FROM ONLY "inbound"."Visits"
NATURAL FULL JOIN "inbound"."InstallationVisits"
NATURAL FULL JOIN "inbound"."SamplingVisits"
NATURAL FULL JOIN "inbound"."PositioningVisits"
NATURAL FULL JOIN "inbound"."OtherVisits"
ORDER BY visit_id ASC
;

CREATE RULE AllVisits_upd0 AS
ON UPDATE TO "inbound"."AllVisits"
DO INSTEAD NOTHING;

GRANT SELECT ON  "inbound"."AllVisits"  TO  viewer_mnmdb;

COMMIT;

```
